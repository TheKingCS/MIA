# M.I.A. Core — v0.1 dependencies
#
# Keep this list minimal. Every dependency added here is a dependency
# every future contributor has to install offline in the field — think
# twice before adding anything heavy (per the project's "avoid
# unnecessary complexity" principle).

PySide6>=6.6,<7.0

# Backup/restore encryption (core/secrets_manager.py). Unlike profile
# password hashing (core/profile_manager.py), which stdlib's hashlib
# covers with no extra dependency, actual data confidentiality has no
# safe stdlib equivalent to reach for — this is the one real exception
# to "keep this list minimal." Has prebuilt wheels for aarch64 (Pi 5).
cryptography>=42,<43

# Reference Library engine (core/reference_library_manager.py, v0.4
# milestone 4.1) — Kiwix's own Python bindings for reading .zim files.
# No equivalent exists in the stdlib and the ZIM format is intentionally
# not reimplemented here (see docs/ROADMAP.md's "don't reinvent the
# wheel" integration decisions). Has prebuilt wheels for aarch64 (Pi 5).
libzim>=3.7,<4.0

# Voice interface (core/voice_manager.py, v0.5 milestone 5.3) — offline
# STT/TTS, no stdlib equivalent. Vosk (Kaldi-based) and Piper (ONNX-based,
# developed by the Rhasspy project specifically targeting Raspberry Pi)
# were chosen over heavier alternatives (e.g. whisper.cpp) because both
# ship prebuilt aarch64 wheels and run on the Pi 5's CPU alone — the AI
# HAT+2's Hailo NPU is an LLM/VLM runtime (hailo-ollama), not a general
# STT/TTS accelerator, per docs/HARDWARE.md.
vosk>=0.3.45,<0.4
piper-tts>=1.4,<2.0

# Mic capture / speaker playback for the Voice interface. Wraps the
# system PortAudio library rather than bundling it on Linux — install
# libportaudio2 via apt (or equivalent) on the target device; see
# core/voice_manager.py's docstring. Not needed for STT/TTS themselves
# (those work from/to .wav files with no audio hardware), only for live
# push-to-talk recording and reply playback.
sounddevice>=0.5,<0.6
numpy>=1.26,<3.0

# Music module's audio tag reading (core/music_manager.py, 2026-09-09)
# — pure Python, prebuilt wheels, no system library. QtMultimedia
# itself (real playback) needs no separate entry here — it ships
# inside the already-pinned PySide6 requirement above. Confirmed
# installable in this dev sandbox.
mutagen>=1.48,<2.0

# Diagnostics System Health panel + Power monitoring (v0.7 milestones
# 7.1/7.2) — cross-platform CPU/RAM/disk/network/battery stats. No
# stdlib equivalent gives this in a cross-platform way; has prebuilt
# wheels for aarch64 (Pi 5).
psutil>=6.0,<8.0

# Navigation module's Sun/Moon calculator (v0.10 milestone 10.2) —
# sunrise/sunset/moon-phase astronomical calculations. Pure Python, no
# network calls, no C extensions — "don't reinvent the wheel" rather
# than hand-rolling potentially-buggy astronomical math.
astral>=3.2,<4.0

# Field Kit's Connected Device Framework (v0.11 milestone 11.1) — serial
# port enumeration (USB-attached MCU boards) for detection/identification.
# Storage/block device detection instead shells out to lsblk directly
# (already present on any Linux target, no Python binding needed).
pyserial>=3.5,<4.0

# Bank/brokerage connectivity (core/plaid_manager.py, 2026-09-08) — the
# official Plaid SDK, over hand-rolled urllib calls (this project's
# usual convention for external APIs, e.g. map tiles). This touches
# live financial credentials; correctness/well-tested auth handling
# matters more here than it did for the stdlib-urllib precedent.
# Version pin confirmed by actually installing it, not guessed.
plaid-python>=44,<45

# Testing
pytest>=7.4
