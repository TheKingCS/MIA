# Backup/restore encryption (core/secrets_manager.py). Unlike profile
# password hashing (core/profile_manager.py), which stdlib's hashlib
# covers with no extra dependency, actual data confidentiality has no
# safe stdlib equivalent to reach for — this is the one real exception
# to "keep this list minimal." Has prebuilt wheels for aarch64 (Pi 5).

# Reference Library engine (core/reference_library_manager.py, v0.4
# milestone 4.1) — Kiwix's own Python bindings for reading .zim files.
# No equivalent exists in the stdlib and the ZIM format is intentionally
# not reimplemented here (see docs/ROADMAP.md's "don't reinvent the
# wheel" integration decisions). Has prebuilt wheels for aarch64 (Pi 5).
libzim>=3.7,<4.0
